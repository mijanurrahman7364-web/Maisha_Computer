<script>
    let selectedProduct = "";

    function openPayment(name, price) {
        selectedProduct = name;
        document.getElementById('orderInfo').innerText = name + " - " + price + "৳";
        document.getElementById('paymentModal').style.display = "block";
    }

    function closeModal() {
        document.getElementById('paymentModal').style.display = "none";
    }

    function submitOrder() {
        const phone = document.getElementById('customerPhone').value;
        const trx = document.getElementById('trxId').value;
        const method = document.getElementById('method').value;

        if (phone === "" || trx === "") {
            alert("সবগুলো তথ্য সঠিকভাবে দিন!");
            return;
        }

        const orderId = "ORD-" + Math.floor(Math.random() * 90000 + 10000);
        
        const newOrder = {
            id: orderId,
            product: selectedProduct,
            phone: phone,
            trx: trx,
            method: method,
            status: "Pending"
        };

        // ডাটাবেজ (localStorage) এ সেভ করা
        let orders = JSON.parse(localStorage.getItem('myOrders')) || [];
        orders.push(newOrder);
        localStorage.setItem('myOrders', JSON.stringify(orders));

        alert("🎉 অভিনন্দন! আপনার অর্ডার সফল হয়েছে।\nঅর্ডার আইডি: " + orderId);
        closeModal();
    }
</script>
