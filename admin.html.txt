<script>
    // আপনার গোপন পাসওয়ার্ড
    const secretCode = "maisha7364";

    function checkPassword() {
        const pass = document.getElementById('adminPass').value;
        if (pass === secretCode) {
            document.getElementById('login-section').style.display = "none";
            document.getElementById('admin-panel').style.display = "block";
            loadOrders();
        } else {
            alert("ভুল পাসওয়ার্ড! আবার চেষ্টা করুন।");
            location.reload(); // পেজ রিফ্রেশ হবে
        }
    }

    function loadOrders() {
        const orderList = document.getElementById('orderList');
        const orders = JSON.parse(localStorage.getItem('myOrders')) || [];
        
        orderList.innerHTML = "";

        orders.forEach((order, index) => {
            const row = `
                <tr>
                    <td>${order.id}</td>
                    <td>${order.product}</td>
                    <td>${order.phone}</td>
                    <td>${order.method.toUpperCase()}</td>
                    <td><code>${order.trx}</code></td>
                    <td class="${order.status === 'Confirmed' ? 'status-confirmed' : 'status-pending'}">${order.status}</td>
                    <td>
                        <button onclick="confirmOrder(${index})" style="background: #2ecc71; color: white; border: none; padding: 5px 10px; cursor: pointer; border-radius: 3px;">Confirm</button>
                        <button onclick="deleteOrder(${index})" style="background: #E63946; color: white; border: none; padding: 5px 10px; cursor: pointer; border-radius: 3px; margin-left: 5px;">Delete</button>
                    </td>
                </tr>
            `;
            orderList.innerHTML += row;
        });
    }

    function confirmOrder(index) {
        let orders = JSON.parse(localStorage.getItem('myOrders'));
        orders[index].status = "Confirmed";
        localStorage.setItem('myOrders', JSON.stringify(orders));
        loadOrders(); // টেবিল আপডেট
    }

    function deleteOrder(index) {
        if(confirm("আপনি কি এই অর্ডারটি মুছে ফেলতে চান?")) {
            let orders = JSON.parse(localStorage.getItem('myOrders'));
            orders.splice(index, 1);
            localStorage.setItem('myOrders', JSON.stringify(orders));
            loadOrders();
        }
    }

    function clearAllOrders() {
        if(confirm("সব অর্ডার মুছে ফেলতে চান? এটি আর ফিরিয়ে আনা যাবে না।")) {
            localStorage.removeItem('myOrders');
            loadOrders();
        }
    }
</script>
